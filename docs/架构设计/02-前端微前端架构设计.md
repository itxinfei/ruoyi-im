## 一、架构概览

前端采用 **主壳 + 业务模块** 的微前端架构，通过 Vite Module Federation 实现模块懒加载和独立部署能力。

```

│                    主应用壳 Shell                           │
│  ┌──────────────────────────────────────────────────────┐  │
│  │  Layout: L1Nav │ L2Panel │ MainContent              │  │
│  └──────────────────────────────────────────────────────┘  │
│                          │                                  │
│              Module Federation (动态加载)                    │
│                          │                                  │
│  ┌─────────┬─────────┬─────────┬─────────┬─────────┐      │
```

#### 悬浮窗状态机 (IM-C-008)

```
┌──────────┐    最小化     ┌──────────┐    退后台    ┌────────────┐
│   FULL   │ ──────────→  │   MINI   │ ──────────→ │ BACKGROUND │
│ (全屏通话)│ ←──────────  │ (圆形头像) │ ←────────── │ (仅保留麦)  │
└──────────┘   点击恢复    └──────────┘   点击恢复   └────────────┘

behavior:
  FULL     → 正常通话 UI，显示远端视频 + 本地小窗
  MINI     → 直径 60px 圆形头像 + 浮动计时 + 静音按钮
             - 可拖拽到屏幕任意位置
             - 点击恢复至 FULL
  BACKGROUND → 完全隐藏通话 UI，仅浮动小圆点 + 计时
             - 麦克风保持连接（纯语音模式）
             - 点击任意处恢复至 FULL

rule:
  - Electron 模式下允许真正脱离窗口（setAlwaysOnTop + 透明窗）
  - 浏览器模式下 Mini 模式仅在当前标签可见时有效（Page Visibility API）
```

### 4.2 EventBus 封装

```typescript
// composables/useEventBus.ts
import { mitt } from 'mitt'
import { EVENTS } from '@/infrastructure/constants'

type EventMap = {
  [EVENTS.MESSAGE_RECEIVED]: Message
  [EVENTS.CALL_INVITE]: CallInviteData
  // ...
}

export const bus = mitt<EventMap>()

export function useEventBus() {
  const on = <K extends keyof EventMap>(key: K, handler: (data: EventMap[K]) => void) => {
    bus.on(key, handler)
    onBeforeUnmount(() => bus.off(key, handler))
  }
  const emit = bus.emit
  return { on, emit }
}
```

### 4.3 WS → EventBus 桥接

```typescript
// infrastructure/ws/initWebSocket.ts
export function initWsBridge() {
  const { on: onWs } = useWebSocket()

  onWs('message', (msg) => bus.emit(EVENTS.MESSAGE_RECEIVED, msg))
  onWs('typing', () => bus.emit(EVENTS.TYPING_START))
  onWs('read', (data) => bus.emit(EVENTS.READ_RECEIPT, data))
  onWs('presence', (data) => bus.emit(data.online ? EVENTS.PRESENCE_ONLINE : EVENTS.PRESENCE_OFFLINE, data))
  onWs('call', (data) => bus.emit(data.signal + '_CALL', data))
}
```

## 五、WebSocket 封装

### 5.1 useWebSocket composable

```typescript
// composables/useWebSocket.ts
interface UseWsReturn {
  connected: Ref<boolean>
  connect: () => void
  disconnect: () => void
  send: (frame: WsFrame) => void
  on: (type: string, handler: Function) => void
  off: (type: string, handler: Function) => void
}

export function useWebSocket(): UseWsReturn {
  let ws: WebSocket | null = null
  const connected = ref(false)
  const handlers: Record<string, Function[]> = {}

  const connect = () => {
    const token = localStorage.getItem('token')
    ws = new WebSocket(`wss://HOST:8443/ws?token=${token}`)

    ws.onopen = () => {
      connected.value = true
      startHeartbeat()
      flushPendingQueue()
    }

    ws.onclose = () => {
      connected.value = false
      scheduleReconnect()
    }

    ws.onmessage = (event) => {
      const frame = JSON.parse(event.data)
      handlers[frame.type]?.forEach(h => h(frame.data))
    }
  }

  // 指数退避重连 (含抖动)
  const scheduleReconnect = () => {
    let attempt = 0
    const tryConnect = () => {
      const delay = Math.min(1000 * 2 ** attempt, 30000) + Math.random() * 2000
      setTimeout(() => { connect(); attempt++ }, delay)
    }
    tryConnect()
  }

  // 心跳 30s
  const startHeartbeat = () => setInterval(() => send({ type: 'PING' }), 30000)

  return { connected, connect, disconnect: () => ws?.close(), send, on, off }
}
```

### 6.2 Electron 桥接

```typescript
// composables/useElectron.ts
export function useElectron() {
  const isElectron = computed(() => !!window.__electronAPI)

  const showNotification = async (title: string, body: string) => {
    if (isElectron.value) {
      return window.__electronAPI.notification.show(title, body)
    }
    // Web 降级: 使用浏览器 Notification API
    if (Notification.permission === 'granted') {
      new Notification(title, { body })
    }
  }

  const setBadge = (count: number) => {
    isElectron.value && window.__electronAPI.tray.setBadge(count)
  }

  const minimize = () => isElectron.value && window.__electronAPI.window.minimize()
  const close = () => isElectron.value && window.__electronAPI.window.close()

  return { isElectron, showNotification, setBadge, minimize, close }
}
```
