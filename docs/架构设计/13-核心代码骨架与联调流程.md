```typescript
// src/store/modules/im/message/index.ts
import { Module } from 'vuex'
import { RootState } from '@/store/index'
import { MessageFullVO, SendMessageCommand } from '@/types/im'
import { sendMessage as apiSendMessage, recallMessage as apiRecallMessage, fetchHistory } from '@/api/im/message'

export interface MessageState {
  currentSessionId: string | null
  messages: MessageFullVO[]          // 当前会话消息
  unreadCount: Record<string, number>// sessionId → unread
  sendingQueue: MessageFullVO[]      // 发送中队列
  msgIdMap: Record<string, number>   // clientMsgId → index (幂等)
}

const messageModule: Module<MessageState, RootState> = {
  namespaced: true,

  state: () => ({
    currentSessionId: null,
    messages: [],
    unreadCount: {},
    sendingQueue: [],
    msgIdMap: {}
  }),

  getters: {
    currentMessages: (state) =>
      state.currentSessionId
        ? state.messages.filter(m => m.sessionId === state.currentSessionId)
        : [],
    currentUnread: (state) =>
      state.currentSessionId ? (state.unreadCount[state.currentSessionId] || 0) : 0
  },

  mutations: {
    SET_CURRENT_SESSION(state, sessionId: string) {
      state.currentSessionId = sessionId
    },
    ADD_MESSAGE(state, msg: MessageFullVO) {
      // 幂等去重
      if (msg.clientMsgId && state.msgIdMap[msg.clientMsgId] !== undefined) {
        const idx = state.msgIdMap[msg.clientMsgId]
        state.messages.splice(idx, 1, msg)
        return
      }
      state.msgIdMap[msg.clientMsgId] = state.messages.push(msg) - 1
    },
    SET_MESSAGES(state, msgs: MessageFullVO[]) {
      state.messages = msgs
    },
    UPSERT_MESSAGE(state, msg: MessageFullVO) {
      const idx = state.messages.findIndex(m => m.id === msg.id)
      if (idx >= 0) {
        state.messages.splice(idx, 1, msg)
      } else {
        state.messages.push(msg)
      }
    },
    UPDATE_MSG_STATUS(state, { msgId, status }: { msgId: string, status: string }) {
      const msg = state.messages.find(m => m.id === msgId)
      if (msg) msg.status = status
    },
    INCREMENT_UNREAD(state, sessionId: string) {
      state.unreadCount[sessionId] = (state.unreadCount[sessionId] || 0) + 1
    },
    CLEAR_UNREAD(state, sessionId: string) {
      state.unreadCount[sessionId] = 0
    },
    ADD_SENDING(state, msg: MessageFullVO) {
      state.sendingQueue.push(msg)
    },
    REMOVE_SENDING(state, clientMsgId: string) {
      const idx = state.sendingQueue.findIndex(m => m.clientMsgId === clientMsgId)
      if (idx >= 0) state.sendingQueue.splice(idx, 1)
    }
  },

  actions: {
    /**
     * 发起发送: 写入 pending 消息 + 入 sendingQueue, 再由 useImWebSocket 监听 EventBus 发送 WS 帧。
     * 不应将 WebSocket 实例放入 Vuex (非序列化,且跨模块副作用)。
     */
    sendMessage({ commit, rootState }, cmd: SendMessageCommand) {
      const clientMsgId = cmd.clientMsgId
      // 1. 插入本地 pending 消息
      const pendingMsg: MessageFullVO = {
        id: `pending-${clientMsgId}`,
        sessionId: cmd.sessionId,
        senderId: rootState.im.user.userId,
        senderName: rootState.im.user.displayName,
        senderAvatar: rootState.im.user.avatar,
        type: cmd.type,
        content: cmd.content,
        clientSeq: cmd.clientSeq,
        clientMsgId,
        status: 'SENDING',
        sendTime: new Date().toISOString(),
        edited: false,
        recallable: true,
        reactions: [],
        attachments: []
      }
      commit('ADD_MESSAGE', pendingMsg)
      commit('ADD_SENDING', pendingMsg)

      // 2. 通知 useImWebSocket 发送 (EventBus 解耦, 不直接持有 WS 实例)
      window.dispatchEvent(new CustomEvent('im:ws:send', { detail: cmd }))
    },

    /**
     * 重发失败消息 (从 sendingQueue 取出重新发送)
     */
    resendMessage({ state, dispatch }, clientMsgId: string) {
      const msg = state.sendingQueue.find(m => m.clientMsgId === clientMsgId)
      if (msg) {
        dispatch('sendMessage', { ...msg, clientMsgId: msg.clientMsgId })
      }
    },

    // WS 推送消息回调
    onMessageReceived({ commit, dispatch }, msg: MessageFullVO) {
      if (msg.clientMsgId) {
        // 用服务端消息替换 pending 消息
        commit('REMOVE_SENDING', msg.clientMsgId)
      }
      commit('ADD_MESSAGE', msg)
      dispatch('imSession/updateLastMessage', msg, { root: true })

      // 非当前会话 → 未读 +1
      if (msg.sessionId !== this.state.currentSessionId) {
        commit('INCREMENT_UNREAD', msg.sessionId)
      }
    },

    // 消息发送成功 ACK
    onMessageAck({ commit }, { clientMsgId, msgId, serverSeq }) {
      commit('REMOVE_SENDING', clientMsgId)
      const msg = this.state.messages.find((m: MessageFullVO) => m.clientMsgId === clientMsgId)
      if (msg) {
        msg.id = msgId
        msg.serverSeq = serverSeq
        msg.status = 'SENT'
      }
    },

    // 历史消息加载
    async loadHistory({ commit, state }, { sessionId, cursor, direction, limit }) {
      const res = await fetchHistory(sessionId, cursor, direction, limit)
      if (res.code === 200) {
        if (direction === 'before') {
          commit('SET_MESSAGES', [...res.data.list, ...state.messages])
        } else {
          commit('SET_MESSAGES', [...state.messages, ...res.data.list])
        }
        return res.data.hasMore
      }
    },

    // 撤回消息
    async recall({ commit }, msgId: string) {
      const res = await apiRecallMessage(msgId)
      if (res.code === 200) {
        commit('UPDATE_MSG_STATUS', { msgId, status: 'RECALLED' })
      }
    }
  }
}

export { messageModule }
```

### 1.3 useImWebSocket Composable (完整版)

```typescript
// src/composables/useImWebSocket.ts
import { ref, onBeforeUnmount } from 'vue'
import { useStore } from 'vuex'
import { ElNotification } from 'element-plus'

export interface WsFrame {
  type: 'MESSAGE' | 'EVENT' | 'CALL' | 'ACK' | 'PING' | 'PONG' | 'SYSTEM'
  action: string
  requestId?: string
  data: Record<string, unknown>
  timestamp: number
  error?: { code: number; message: string }
}

export function useImWebSocket() {
  const store = useStore()
  const connected = ref(false)
  const ws = ref<WebSocket | null>(null)
  const reconnectCount = ref(0)
  const MAX_RECONNECT = 5
  let heartbeatTimer: number | null = null
  let reconnectTimer: number | null = null

  // ---- 鉴权 Token ----
  function getToken(): string {
    return sessionStorage.getItem('accessToken') || ''
  }

  // ---- 连接建立 ----
  function connect() {
    if (ws.value?.readyState <= WebSocket.OPEN) return

    const url = `${import.meta.env.VITE_WS_URL || 'wss://localhost'}/im/ws?token=${getToken()}`
    ws.value = new WebSocket(url)

    ws.value.onopen = () => {
      console.log('[IM-WS] connected')
      connected.value = true
      reconnectCount.value = 0
      store.dispatch('im/wsConnected')
      startHeartbeat()
    }

    ws.value.onmessage = (e: MessageEvent) => {
      try {
        const frame: WsFrame = JSON.parse(e.data)
        handleFrame(frame)
      } catch (err) {
        console.error('[IM-WS] parse error', err)
      }
    }

    ws.value.onclose = (e: CloseEvent) => {
      console.warn('[IM-WS] closed', e.code, e.reason)
      connected.value = false
      stopHeartbeat()
      store.dispatch('im/wsDisconnected')

      if (e.code !== 1000 && e.code !== 1001) {
        scheduleReconnect()
      }
    }

    ws.value.onerror = () => {
      console.error('[IM-WS] error')
    }
  }

  // ---- 帧分发 ----
  function handleFrame(frame: WsFrame) {
    switch (frame.type) {
      case 'MESSAGE':
        store.dispatch('im/message/onMessageReceived', frame.data, { root: true })
        break
      case 'ACK':
        if (frame.action === 'message.ack') {
          store.dispatch('im/message/onMessageAck', frame.data, { root: true })
        }
        break
      case 'EVENT':
        handleEvent(frame)
        break
      case 'CALL':
        store.dispatch('im/call/onCallSignal', frame.data, { root: true })
        break
      case 'PONG':
        // 心跳回应, 无需处理
        break
      case 'SYSTEM':
        handleSystem(frame)
        break
    }
  }

  // ---- 事件处理 ----
  function handleEvent(frame: WsFrame) {
    switch (frame.action) {
      case 'user.presence':
        store.dispatch('im/contact/updatePresence', frame.data, { root: true })
        break
      case 'session.update':
        store.dispatch('im/session/updateSession', frame.data, { root: true })
        break
      case 'message.typing':
        store.dispatch('im/session/setTyping', frame.data, { root: true })
        break
      case 'group.member_change':
        store.dispatch('im/group/onMemberChange', frame.data, { root: true })
        break
    }
  }

  // ---- 系统通知 ----
  function handleSystem(frame: WsFrame) {
    switch (frame.action) {
      case 'notification.new':
        ElNotification({
          title: frame.data.title as string,
          message: frame.data.body as string,
          type: 'info',
          duration: 5000
        })
        break
      case 'server.restarting':
        ElNotification({
          title: '服务器维护',
          message: '服务器即将重启, 请保存工作内容',
          type: 'warning',
          duration: 0
        })
        break
      case 'token.expired':
        // Token 过期, 用 Refresh Token 刷新
        break
    }
  }

  // ---- 心跳 ----
  function startHeartbeat() {
    heartbeatTimer = window.setInterval(() => {
      if (ws.value?.readyState === WebSocket.OPEN) {
        ws.value.send(JSON.stringify({ type: 'PING', timestamp: Date.now() }))
      }
    }, 30_000)
  }

  function stopHeartbeat() {
    if (heartbeatTimer !== null) {
      clearInterval(heartbeatTimer)
      heartbeatTimer = null
    }
  }

  // ---- 重连 (指数退避) ----
  function scheduleReconnect() {
    if (reconnectCount.value >= MAX_RECONNECT) {
      ElNotification({
        title: '离线',
        message: '服务器连接失败, 请检查网络后重试',
        type: 'error',
        duration: 0
      })
      return
    }
    const delay = Math.min(1000 * 2 ** reconnectCount.value, 30_000)
    reconnectTimer = window.setTimeout(() => {
      reconnectCount.value++
      console.log(`[IM-WS] reconnecting #${reconnectCount.value}`)
      connect()
    }, delay)
  }

  // ---- 主动断开 ----
  function disconnect() {
    if (reconnectTimer !== null) clearTimeout(reconnectTimer)
    stopHeartbeat()
    if (ws.value) {
      ws.value.close(1000, '主动断开')
      ws.value = null
    }
  }

  // ---- 发送消息封装 ----
  function send(frame: Omit<WsFrame, 'timestamp'>) {
    if (!ws.value || ws.value.readyState !== WebSocket.OPEN) {
      throw new Error('WebSocket 未连接')
    }
    ws.value.send(JSON.stringify({ ...frame, timestamp: Date.now() }))
  }

  // ---- 生命周期 ----
  onBeforeUnmount(() => {
    disconnect()
  })

  return { ws, connected, connect, disconnect, send, reconnectCount }
}
```

```java
    private Integer memberCount = 0;

    public void addMember(Long operatorId, Long newMemberId) {
        if (this.status.equals("DISMISSED")) {
            throw new DomainException("群组已解散");
        }
        if (this.memberCount >= this.maxMemberCount) {
            throw new DomainException("群成员已满 (max=" + this.maxMemberCount + ")");
        }
        this.memberCount++;
    }

    public void removeMember(Long operatorId, Long memberId) {
        if (this.memberId.equals(this.ownerId)) {
            throw new DomainException("不能移除群主");
        }
        if (this.memberCount <= 0) {
            throw new DomainException("群成员数为 0");
        }
        this.memberCount--;
    }

    public void transferOwnership(Long oldOwnerId, Long newOwnerId) {
        if (!this.ownerId.equals(oldOwnerId)) {
            throw new DomainException("非群主无权转让");
        }
        this.ownerId = newOwnerId;
    }

    public void dismiss(Long operatorId) {
        if (!this.ownerId.equals(operatorId)) {
            throw new DomainException("仅群主可解散群组");
        }
        this.status = "DISMISSED";
    }

    public void updateProfile(Long operatorId, String newName, String newDesc, String newAvatar) {
        if (!hasPermission(operatorId, "group:update")) {
            throw new DomainException("无权修改群信息");
        }
        if (StrUtil.isNotBlank(newName)) this.name = newName;
        if (newDesc != null) this.description = newDesc;
        if (StrUtil.isNotBlank(newAvatar)) this.avatar = newAvatar;
    }

    private boolean hasPermission(Long userId, String permission) {
        return this.ownerId.equals(userId) || "group:admin".equals(permission);
    }
```

```java
// application/GroupApplicationService.java
@Service
@RequiredArgsConstructor
public class GroupApplicationService {

    private final GroupRepository groupRepository;
    private final GroupDomainService domainService;
    private final GroupMemberRepository memberRepository;
    private final ApplicationEventPublisher eventPublisher;
    private final UserServiceClient userServiceClient;
    private final GroupConverter converter;

    @Transactional(rollbackFor = Exception.class)
    public GroupDetailVO createGroup(Long operatorId, CreateGroupCommand cmd) {
        // 1. 领域规则校验
        domainService.validateGroupCreation(cmd, groupRepository);

        // 2. 构建聚合根
        Group group = new Group();
        group.setName(cmd.getName());
        group.setAvatar(cmd.getAvatar());
        group.setDescription(cmd.getDescription());
        group.setOwnerId(operatorId);
        group.setGroupType("NORMAL");
        group.setStatus("ACTIVE");
        group.setMemberCount(cmd.getMemberIds().size() + 1);  // 群主 + 初始成员
        group.setMaxMemberCount(500);
        group.setCreateBy(operatorId);
        group.setTags(JSONUtil.toJsonStr(cmd.getTags()));

        // 3. 持久化群
        groupRepository.save(group);

        // 4. 批量插入群成员
        List<GroupMember> members = new ArrayList<>();
        members.add(createGroupMember(group.getId(), operatorId, GroupRole.OWNER));  // 群主
        cmd.getMemberIds().forEach(uid ->
            members.add(createGroupMember(group.getId(), uid, GroupRole.MEMBER))
        );
        memberRepository.saveBatch(members);

        // 5. 发布领域事件
        eventPublisher.publishEvent(new GroupCreatedEvent(
            group.getId(), group.getName(), operatorId, cmd.getMemberIds(),
            LocalDateTime.now()
        ));

        // 6. 返回 VO
        return converter.toDetailVO(group, operatorId);
    }

    @Transactional(rollbackFor = Exception.class)
    public void addMember(Long groupId, Long operatorId, List<Long> userIds) {
        Group group = groupRepository.getById(groupId);

        // 校验操作者是否为群主/管理员 (成员校验可通过群成员查询, 此处简化)
        ensurePermission(group, operatorId, "group:add_member");

        // 逐个添加
        for (Long uid : userIds) {
            domainService.ensureCanAddMember(group, uid);
            if (memberRepository.exists(group.getId(), uid)) {
                continue;
            }
            group.addMember(operatorId, uid);
            memberRepository.save(createGroupMember(group.getId(), uid, GroupRole.MEMBER));
        }

        groupRepository.save(group);  // 更新 memberCount 等
    }

    @Transactional(rollbackFor = Exception.class)
    public void updateGroup(Long groupId, Long operatorId, UpdateGroupCommand cmd) {
        Group group = groupRepository.getById(groupId);
        group.updateProfile(operatorId, cmd.getName(), cmd.getDescription(), cmd.getAvatar());
        groupRepository.save(group);
    }

    private boolean ensurePermission(Group group, Long userId, String perm) {
        if (group.getOwnerId().equals(userId)) return true;
        GroupMember member = memberRepository.findByGroupAndUser(group.getId(), userId);
        if (member != null && member.getRole().equals("ADMIN")) return true;
        throw new DomainException("无权执行此操作");
    }

    private GroupMember createGroupMember(Long groupId, Long userId, GroupRole role) {
        GroupMember gm = new GroupMember();
        gm.setGroupId(groupId);
        gm.setUserId(userId);
        gm.setRole(role.getCode());
        gm.setJoinTime(LocalDateTime.now());
        gm.setJoinType("INVITED");
        gm.setDelFlag(0);
        return gm;
    }
}
```

```java
    @Override
    public void save(Group group) {
        if (group.getId() == null) {
            GroupPO po = converter.toPO(group);
            groupMapper.insert(po);
        } else {
            GroupPO po = converter.toPO(group);
            groupMapper.updateById(po);
        }
    }

    @Override
    public Optional<Group> findById(Long id) {
        GroupPO po = groupMapper.selectById(id);
        return Optional.ofNullable(converter.toEntity(po));
    }

    @Override
    public Group getById(Long id) {
        Optional<Group> opt = findById(id);
        return opt.orElseThrow(() -> new DomainException("群不存在 (id=" + id + ")"));
    }

    @Override
    public List<Group> findByIds(Collection<Long> ids) {
        if (CollUtil.isEmpty(ids)) return Collections.emptyList();
        return groupMapper.selectBatchIds(ids).stream()
            .map(converter::toEntity)
            .collect(Collectors.toList());
    }

    @Override
    public boolean existsByName(String name) {
        return groupMapper.exists(new QueryWrapper<GroupPO>().eq("name", name));
    }

    @Override
    public List<Group> findByMemberId(Long memberId, int page, int size) {
        int offset = (page - 1) * size;
        return groupMapper.selectGroupsByMember(memberId, offset, size).stream()
            .map(converter::toEntity)
            .collect(Collectors.toList());
    }
}
```

## 八、axois 请求拦截器

```javascript
// src/utils/request.js
import axios from 'axios'
import { ElMessage, ElMessageBox } from 'element-plus'
import { useStore } from 'vuex'

const service = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || '/im/',
  timeout: 15000
})

// 请求拦截器
service.interceptors.request.use(
  config => {
    const token = sessionStorage.getItem('accessToken')
    if (token) {
      config.headers['Authorization'] = `Bearer ${token}`
    }
    config.headers['Content-Type'] = 'application/json;charset=UTF-8'
    config.headers['X-Request-Id'] = `req-${Date.now()}`
    return config
  },
  error => Promise.reject(error)
)

// 响应拦截器
service.interceptors.response.use(
  response => {
    const res = response.data
    if (res.code === 200) return res
    if (res.code === 30002) {
      // Token 过期 → 尝试刷新
      return handleTokenRefresh(response.config, res)
    }
    ElMessage.error(res.msg || '请求失败')
    return Promise.reject(new Error(res.msg || 'Error'))
  },
  error => {
    if (error.response?.status === 401) {
      ElMessageBox.confirm('登录已过期, 是否重新登录?', '提示', { type: 'warning' })
        .then(() => location.href = '/login')
    } else if (error.code === 'ECONNABORTED') {
      ElMessage.error('请求超时')
    } else {
      ElMessage.error(error.message || '网络错误')
    }
    return Promise.reject(error)
  }
)

let isRefreshing = false
let requestsQueue = []

async function handleTokenRefresh(failedConfig, errorRes) {
  if (!isRefreshing) {
    isRefreshing = true
    try {
      const refreshToken = localStorage.getItem('refreshToken')
      const res = await axios.post('/im/auth/refresh', { refreshToken })
      if (res.data.code === 200) {
        const { accessToken, refreshToken: newRefresh } = res.data.data
        sessionStorage.setItem('accessToken', accessToken)
        localStorage.setItem('refreshToken', newRefresh)
        // 重试排队的请求
        requestsQueue.forEach(cb => cb(accessToken))
        requestsQueue = []
        failedConfig.headers['Authorization'] = `Bearer ${accessToken}`
        return service(failedConfig)
      }
    } catch {
      location.href = '/login'
    } finally {
      isRefreshing = false
    }
  }
  // 正在刷新 → 排队
  return new Promise(resolve => {
    requestsQueue.push(token => {
      failedConfig.headers['Authorization'] = `Bearer ${token}`
      resolve(service(failedConfig))
    })
  })
}

export default service
```
