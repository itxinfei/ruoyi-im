## 一、异常类体系

```java
// common/exception/BaseException.java
@Getter
public class BaseException extends RuntimeException {
    private final Integer errorCode;
    private final String userMessage;

    public BaseException(Integer errorCode, String userMessage) {
        super(userMessage);
        this.errorCode = errorCode;
        this.userMessage = userMessage;
    }

    public BaseException(Integer errorCode, String message, Throwable cause) {
        super(message, cause);
        this.errorCode = errorCode;
        this.userMessage = message;
    }
}

// common/exception/ParamException.java —— 参数校验异常
public class ParamException extends BaseException {
    private List<FieldError> fieldErrors;

    public ParamException(String msg) {
        super(ErrorCode.PARAM_ERROR, msg);
    }

    public ParamException(List<FieldError> fieldErrors) {
        super(ErrorCode.PARAM_ERROR, "参数校验失败");
        this.fieldErrors = fieldErrors;
    }

    @Data
    @AllArgsConstructor
    public static class FieldError {
        private String field;
        private String message;
    }
}

// common/exception/AuthException.java —— 认证异常
public class AuthException extends BaseException {
    public AuthException(String msg) {
        super(ErrorCode.AUTH_FAIL, msg);
    }
    public AuthException(Integer code, String msg) {
        super(code, msg);
    }
}

// common/exception/ForbiddenException.java —— 权限异常
public class ForbiddenException extends BaseException {
    public ForbiddenException(String msg) {
        super(ErrorCode.PERMISSION_DENIED, msg);
    }
    public ForbiddenException(String permission, String resource) {
        super(ErrorCode.PERMISSION_DENIED,
              "无权执行此操作, 需要 " + permission + " 权限 (资源: " + resource + ")");
    }
}

// common/exception/ResourceNotFoundException.java —— 资源不存在
public class ResourceNotFoundException extends BaseException {
    public ResourceNotFoundException(String resourceType, Object id) {
        super(ErrorCode.USER_NOT_FOUND, resourceType + " 不存在 (id=" + id + ")");
    }
}

// common/exception/BusinessException.java —— 通用业务异常
public class BusinessException extends BaseException {
    public BusinessException(Integer code, String msg) {
        super(code, msg);
    }
}

// common/exception/RateLimitException.java —— 限流
public class RateLimitException extends BaseException {
    private final int retryAfterSeconds;

    public RateLimitException(int retryAfterSeconds) {
        super(ErrorCode.RATE_LIMIT, "请求过于频繁, 请 " + retryAfterSeconds + " 秒后重试");
        this.retryAfterSeconds = retryAfterSeconds;
    }

    public int getRetryAfterSeconds() { return retryAfterSeconds; }
}
```

## 二、全局异常处理器

```java
// common/exception/GlobalExceptionHandler.java —— 统一异常处理器

@RestControllerAdvice
@Slf4j
public class GlobalExceptionHandler {

    // ========== 参数校验 ==========

    /**
     * @RequestBody @Valid 校验失败
     */
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public Result<Map<String, Object>> handleValidation(MethodArgumentNotValidException ex,
                                                          HttpServletRequest req) {
        List<Map<String, String>> fieldErrors = ex.getBindingResult().getFieldErrors().stream()
            .map(e -> Map.of("field", e.getField(), "message", e.getDefaultMessage()))
            .collect(Collectors.toList());

        Map<String, Object> data = Map.of(
            "fieldErrors", fieldErrors
        );

        log.warn("[参数校验失败] uri={}, errors={}", req.getRequestURI(), fieldErrors);
        return Result.fail(ErrorCode.PARAM_ERROR, "参数校验失败");
    }

    /**
     * @RequestParam / @PathVariable 校验失败
     */
    @ExceptionHandler(ConstraintViolationException.class)
    public Result<Void> handleConstraintViolation(ConstraintViolationException ex,
                                                   HttpServletRequest req) {
        log.warn("[约束校验失败] uri={}, msg={}", req.getRequestURI(), ex.getMessage());
        return Result.fail(ErrorCode.PARAM_ERROR, ex.getMessage());
    }

    /**
     * 参数类型转换错误
     */
    @ExceptionHandler(BindException.class)
    public Result<Void> handleBindException(BindException ex,
                                             HttpServletRequest req) {
        log.warn("[类型绑定错误] uri={}, msg={}", req.getRequestURI(), ex.getMessage());
        return Result.fail(ErrorCode.PARAM_ERROR, "参数类型错误");
    }

    /**
     * JSON 格式错误
     */
    @ExceptionHandler(HttpMessageNotReadableException.class)
    public Result<Void> handleJsonError(HttpMessageNotReadableException ex,
                                         HttpServletRequest req) {
        log.warn("[JSON解析错误] uri={}, msg={}", req.getRequestURI(), ex.getMessage());
        return Result.fail(ErrorCode.PARAM_ERROR, "请求体格式错误");
    }

    /**
     * 缺少参数
     */
    @ExceptionHandler(MissingServletRequestParameterException.class)
    public Result<Void> handleMissingParam(MissingServletRequestParameterException ex) {
        return Result.fail(ErrorCode.PARAM_ERROR, "缺少参数: " + ex.getParameterName());
    }

    /**
     * 参数类型不匹配
     */
    @ExceptionHandler(MethodArgumentTypeMismatchException.class)
    public Result<Void> handleTypeMismatch(MethodArgumentTypeMismatchException ex) {
        return Result.fail(ErrorCode.PARAM_ERROR,
            "参数 " + ex.getName() + " 类型不匹配");
    }

    /**
     * Spring 自带参数校验
     */
    @ExceptionHandler(IllegalArgumentException.class)
    public Result<Void> handleIllegalArgument(IllegalArgumentException ex) {
        return Result.fail(ErrorCode.PARAM_ERROR, ex.getMessage());
    }

    // ========== 自定义业务异常 ==========

    @ExceptionHandler(ParamException.class)
    public Result<Map<String, Object>> handleParam(ParamException ex) {
        Map<String, Object> data = CollUtil.isNotEmpty(ex.getFieldErrors())
            ? Map.of("fieldErrors",
                ex.getFieldErrors().stream()
                    .map(fe -> Map.of("field", fe.getField(), "message", fe.getMessage()))
                    .collect(Collectors.toList()))
            : null;
        return Result.fail(ex.getErrorCode(), ex.getUserMessage());
    }

    @ExceptionHandler(AuthException.class)
    public Result<Void> handleAuth(AuthException ex) {
        log.warn("[认证异常] {}", ex.getMessage());
        return Result.fail(ex.getErrorCode(), ex.getUserMessage());
    }

    @ExceptionHandler(ForbiddenException.class)
    public Result<Void> handleForbidden(ForbiddenException ex) {
        log.warn("[权限不足] {}", ex.getMessage());
        return Result.fail(ex.getErrorCode(), ex.getUserMessage());
    }

    @ExceptionHandler(ResourceNotFoundException.class)
    public Result<Void> handleNotFound(ResourceNotFoundException ex) {
        log.warn("[资源不存在] {}", ex.getMessage());
        return Result.fail(ex.getErrorCode(), ex.getUserMessage());
    }

    @ExceptionHandler(BusinessException.class)
    public Result<Void> handleBusiness(BusinessException ex) {
        log.warn("[业务异常] code={}, msg={}", ex.getErrorCode(), ex.getMessage());
        return Result.fail(ex.getErrorCode(), ex.getUserMessage());
    }

    @ExceptionHandler(RateLimitException.class)
    public ResponseEntity<Result<Void>> handleRateLimit(RateLimitException ex) {
        HttpHeaders headers = new HttpHeaders();
        headers.add("Retry-After", String.valueOf(ex.getRetryAfterSeconds()));
        return ResponseEntity.status(HttpStatus.TOO_MANY_REQUESTS)
            .headers(headers)
            .body(Result.fail(ex.getErrorCode(), ex.getUserMessage()));
    }

    // ========== 数据访问异常 ==========

    /**
     * 唯一键冲突
     */
    @ExceptionHandler(DuplicateKeyException.class)
    public Result<Void> handleDuplicateKey(DuplicateKeyException ex) {
        log.warn("[唯一键冲突] {}", ex.getMessage());
        return Result.fail(ErrorCode.USER_ALREADY_EXISTS, "数据已存在");
    }

    /**
     * 数据库连接异常
     */
    @ExceptionHandler({DataAccessResourceFailureException.class,
                       CannotGetJdbcConnectionException.class})
    public Result<Void> handleDatabaseDown(Exception ex) {
        log.error("[数据库连接异常]", ex);
        return Result.fail(ErrorCode.SYSTEM_ERROR, "数据库服务暂时不可用");
    }

    /**
     * 乐观锁冲突
     */
    @ExceptionHandler(OptimisticLockException.class)
    public Result<Void> handleOptimisticLock(OptimisticLockException ex) {
        return Result.fail(ErrorCode.SYSTEM_ERROR, "数据已被修改, 请刷新后重试");
    }

    // ========== HTTP 通用异常 ==========

    /**
     * 不支持的 HTTP 方法
     */
    @ExceptionHandler(HttpRequestMethodNotSupportedException.class)
    public Result<Void> handleMethodNotSupported(HttpRequestMethodNotSupportedException ex) {
        return Result.fail(405, "不支持的请求方法: " + ex.getMethod());
    }

    /**
     * 不支持的 Content-Type
     */
    @ExceptionHandler(HttpMediaTypeNotSupportedException.class)
    public Result<Void> handleMediaTypeNotSupported(HttpMediaTypeNotSupportedException ex) {
        return Result.fail(415, "不支持的 Content-Type: " + ex.getContentType());
    }

    /**
     * 不支持的 Accept
     */
    @ExceptionHandler(HttpMediaTypeNotAcceptableException.class)
    public Result<Void> handleMediaTypeNotAcceptable() {
        return Result.fail(406, "不支持的 Accept 媒体类型");
    }

    // ========== 文件上传 ==========

    @ExceptionHandler(MultipartException.class)
    public Result<Void> handleMultipart(MultipartException ex) {
        log.warn("[文件上传错误] {}", ex.getMessage());
        return Result.fail(ErrorCode.UPLOAD_FAILED, "文件上传失败");
    }

    @ExceptionHandler(MaxUploadSizeExceededException.class)
    public Result<Void> handleUploadSize() {
        return Result.fail(ErrorCode.FILE_TOO_LARGE, "文件大小超过限制");
    }

    // ========== Feign 调用异常 ==========

    @ExceptionHandler(FeignException.class)
    public Result<Void> handleFeignException(FeignException ex) {
        log.error("[Feign 调用异常] status={}, msg={}", ex.status(), ex.getMessage());
        return Result.fail(ErrorCode.SYSTEM_ERROR, "服务间调用失败");
    }

    // ========== 兜底 ==========

    /**
     * 捕获所有未处理的异常
     */
    @ExceptionHandler(Exception.class)
    public Result<Void> handleAll(Exception ex, HttpServletRequest req) {
        log.error("[未处理异常] uri={}", req.getRequestURI(), ex);
        return Result.fail(ErrorCode.SYSTEM_ERROR, "系统异常, 请稍后重试");
    }

    /**
     * Throwable 兜底
     */
    @ExceptionHandler(Throwable.class)
    public Result<Void> handleThrowable(Throwable ex, HttpServletRequest req) {
        log.error("[JVM 异常] uri={}", req.getRequestURI(), ex);
        return Result.fail(ErrorCode.SYSTEM_ERROR, "严重异常, 请联系管理员");
    }
}
```

## 五、开发规范:何时抛出何种异常

```java
// ✅ 正确: 在 DomainService 抛出带业务语义的异常
public void addMember(Group group, Long newMemberId) {
    if (group.getMemberCount() >= group.getMaxMemberCount()) {
        throw new BusinessException(ErrorCode.GROUP_FULL,
            "群成员已满 (max=" + group.getMaxMemberCount() + ")");
    }
}

// ✅ 正确: Application 层校验参数后抛出
public GroupDetailVO createGroup(Long operatorId, CreateGroupCommand cmd) {
    if (StrUtil.isBlank(cmd.getName())) {
        throw new ParamException("群名称不能为空");
    }
    // ...
}

// ✅ 正确: 调用其他服务结果为空时
UserDTO user = userServiceClient.getSimple(userId);
if (user == null) {
    throw new ResourceNotFoundException("用户", userId);
}

// ✅ 正确: 权限校验
if (!group.getOwnerId().equals(operatorId)) {
    throw new ForbiddenException("group:update", "group:" + group.getId());
}

// ✅ 正确: Auth 层抛出
if (!password.matches("^(?=.*[a-z])(?=.*[A-Z])(?=.*\\d).{8,20}$")) {
    throw new BusinessException(ErrorCode.PARAM_ERROR, "密码需含大小写字母和数字, 长度 8-20");
}

// ✅ 正确: 限流检查
if (!rateLimiter.tryAcquire("user:" + userId)) {
    throw new RateLimitException(60);
}

// ❌ 错误: 不要捕获所有异常后吞掉
try {
    riskyOperation();
} catch (Exception e) {
    // 绝对不能空 catch!
    log.error("异常", e);
    throw new BusinessException(ErrorCode.SYSTEM_ERROR, "操作失败");  // 重新抛出
}
```

## 六、常见异常场景处理示例

### 6.1 消息撤回超时

```java
// Domain Service
public void recallMessage(Message msg, Long operatorId) {
    // 1. 权限校验
    if (!msg.getSenderId().equals(operatorId)) {
        throw new ForbiddenException("msg:recall", "message:" + msg.getId());
    }
    // 2. 时间窗口校验
    if (Duration.between(msg.getSendTime(), LocalDateTime.now()).toMinutes()
            > MAX_RECALL_WINDOW_MINUTES) {
        throw new BusinessException(ErrorCode.MSG_RECALL_TIMEOUT,
            "消息发送超过 " + MAX_RECALL_WINDOW_MINUTES + " 分钟, 不可撤回");
    }
    // 3. 执行撤回
    msg.recall();
}
```
